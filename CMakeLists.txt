cmake_minimum_required(VERSION 3.10)
project(variable_thrust_trajectory C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(trajectory_core c/trajectory.c)
target_include_directories(trajectory_core PUBLIC c)
if(NOT MSVC)
  target_link_libraries(trajectory_core PUBLIC m)
endif()
add_executable(trajectory_demo c/main.c)
target_link_libraries(trajectory_demo PRIVATE trajectory_core)
add_executable(check_core tests/check_core.c)
target_link_libraries(check_core PRIVATE trajectory_core)
enable_testing()
add_test(NAME analytic_physics COMMAND check_core)

# Independent six-DOF subsystem; original three-DOF targets remain available.
add_library(six_dof_core six_dof/c/six_dof.c)
target_include_directories(six_dof_core PUBLIC six_dof/c)
if(NOT MSVC)
  target_link_libraries(six_dof_core PUBLIC m)
endif()
add_executable(six_dof_demo six_dof/c/main.c)
target_link_libraries(six_dof_demo PRIVATE six_dof_core)
add_executable(check_six_dof_core six_dof/tests/check_core.c)
target_link_libraries(check_six_dof_core PRIVATE six_dof_core)
add_executable(check_six_dof_events six_dof/tests/check_events.c)
target_link_libraries(check_six_dof_events PRIVATE six_dof_core)
add_test(NAME six_dof_analytic_physics COMMAND check_six_dof_core)
add_test(NAME six_dof_events COMMAND check_six_dof_events)

# Solid booster and conditional cruise ignition using the unchanged six-DOF RHS.
add_library(boost_cruise_core six_dof/boost_cruise/c/boost_cruise.c)
target_include_directories(boost_cruise_core PUBLIC six_dof/boost_cruise/c)
target_link_libraries(boost_cruise_core PUBLIC six_dof_core)
add_executable(boost_cruise_demo six_dof/boost_cruise/c/main.c)
target_link_libraries(boost_cruise_demo PRIVATE boost_cruise_core)
add_executable(check_boost_cruise_model six_dof/boost_cruise/tests/check_model.c)
target_link_libraries(check_boost_cruise_model PRIVATE boost_cruise_core)
add_test(NAME boost_cruise_model_events COMMAND check_boost_cruise_model)

# Civil fixed-route controllers and independent traffic safety supervision.
add_library(civil_route_core six_dof/civil_route/c/controllers.c six_dof/civil_route/c/civil_route.c)
target_include_directories(civil_route_core PUBLIC six_dof/civil_route/c)
target_link_libraries(civil_route_core PUBLIC boost_cruise_core)
add_executable(civil_route_demo six_dof/civil_route/c/main.c)
target_link_libraries(civil_route_demo PRIVATE civil_route_core)
add_executable(check_civil_route_components six_dof/civil_route/tests/check_components.c)
target_link_libraries(check_civil_route_components PRIVATE civil_route_core)
add_test(NAME civil_route_controllers_traffic COMMAND check_civil_route_components)
