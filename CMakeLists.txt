cmake_minimum_required(VERSION 3.10)
project(variable_thrust_trajectory C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
add_library(trajectory_core c/trajectory.c)
target_include_directories(trajectory_core PUBLIC c)
if(NOT MSVC)
  target_link_libraries(trajectory_core PUBLIC m)
endif()
add_executable(trajectory_demo c/main.c)
target_link_libraries(trajectory_demo PRIVATE trajectory_core)
add_executable(check_core tests/check_core.c)
target_link_libraries(check_core PRIVATE trajectory_core)
enable_testing()
add_test(NAME analytic_physics COMMAND check_core)

# Independent six-DOF subsystem; original three-DOF targets remain available.
add_library(six_dof_core six_dof/c/six_dof.c)
target_include_directories(six_dof_core PUBLIC six_dof/c)
if(NOT MSVC)
  target_link_libraries(six_dof_core PUBLIC m)
endif()
add_executable(six_dof_demo six_dof/c/main.c)
target_link_libraries(six_dof_demo PRIVATE six_dof_core)
add_executable(check_six_dof_core six_dof/tests/check_core.c)
target_link_libraries(check_six_dof_core PRIVATE six_dof_core)
add_executable(check_six_dof_events six_dof/tests/check_events.c)
target_link_libraries(check_six_dof_events PRIVATE six_dof_core)
add_test(NAME six_dof_analytic_physics COMMAND check_six_dof_core)
add_test(NAME six_dof_events COMMAND check_six_dof_events)
